# Пример про Зенит и Спартак (по мотивам примера 1 из лекции 11.10.2024)
# Z - выигрывает Зенит, S - выигрывает Спартак, P - торжествует Питер, M - торжествует Москва
# число аксиом
4
# если выигрывает Зенит, то Питер торжествует
Z -> P
# если выигрывает Спартак, то Москва торжествует
S -> M
# выигрывает Зенит или Спартак
Z v S
# если Питер торжествует, то Москва не торжествует
P -> ~M
# цель: Питер торжествует тогда и только тогда, когда не торжествует Москва
P <-> ~M
