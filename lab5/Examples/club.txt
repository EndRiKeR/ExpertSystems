# Пример из лекции 01.11.2024: члены клуба Петя, Коля, Лена
# S(x) - x лыжник, M(x) - x альпинист, L(x, y) - x любит y
# число дизъюнктов аксиом
7
# каждый член клуба лыжник или альпинист
S(x1) v M(x1)
# нет альпиниста, который любит дождь
~M(x2) v ~L(x2, Дождь)
# все лыжники любят снег
~S(x3) v L(x3, Снег)
# Лена не любит то, что любит Петя, и любит то, что Петя не любит
~L(Лена, y1) v ~L(Петя, y1)
L(Лена, y2) v L(Петя, y2)
# Петя любит и дождь, и снег
L(Петя, Дождь)
L(Петя, Снег)
# число дизъюнктов отрицания цели
1
# цель: есть альпинист, не являющийся лыжником; ее отрицание:
~M(x4) v S(x4)
